local Connections = {}

local function connect(name, signal, fn)
    if Connections[name] then Connections[name]:Disconnect() end
    Connections[name] = signal:Connect(fn)
end

local function disconnectNamed(name)
    if Connections[name] then
        Connections[name]:Disconnect()
        Connections[name] = nil
    end
end

local function disconnectAll()
    for _, conn in pairs(Connections) do
        pcall(function() conn:Disconnect() end)
    end
end

local function tween(instance, duration, props, style, direction)
    style     = style     or Enum.EasingStyle.Quad
    direction = direction or Enum.EasingDirection.Out
    TweenService:Create(instance, TweenInfo.new(duration, style, direction), props):Play()
end

local function applyCorner(parent, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 5)
    c.Parent       = parent
    return c
end

local function applyPadding(parent, top, left, right, bottom)
    local p = Instance.new("UIPadding")
    p.PaddingTop    = UDim.new(0, top    or 0)
    p.PaddingLeft   = UDim.new(0, left   or 0)
    p.PaddingRight  = UDim.new(0, right  or 0)
    p.PaddingBottom = UDim.new(0, bottom or 0)
    p.Parent        = parent
    return p
end

local function applyListLayout(parent, spacing, sortOrder)
    local l = Instance.new("UIListLayout")
    l.Padding   = UDim.new(0, spacing or 5)
    l.SortOrder = sortOrder or Enum.SortOrder.LayoutOrder
    l.Parent    = parent
    return l
end

local function setParticles(enabled)
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("ParticleEmitter") or v:IsA("Fire")
        or v:IsA("Smoke")          or v:IsA("Sparkles") then
            v.Enabled = enabled
        end
    end
end

    if activeDrag and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
        activeDrag(input.Position.X)
    end

    if input.UserInputType == Enum.UserInputType.MouseButton1

    row.Size             = UDim2.new(1, -4, 0, 28)
    row.BackgroundColor3 = Theme.BgRow
    row.BorderSizePixel  = 0
    row.Parent           = panel
    applyCorner(row, 5)

    lbl.Text              = label
    lbl.Size              = UDim2.new(1, -56, 1, 0)
    lbl.Position          = UDim2.new(0, 8, 0, 0)

    lbl.TextColor3        = Theme.TextSecondary
    lbl.Font              = Enum.Font.Gotham
    lbl.TextSize          = 12
    lbl.TextXAlignment    = Enum.TextXAlignment.Left
    lbl.Parent            = row

    local track = Instance.new("Frame")
    track.Size             = UDim2.new(0, 38, 0, 18)
    track.Position         = UDim2.new(1, -46, 0.5, -9)
    track.BackgroundColor3 = state and Theme.AccentTrackOn or Theme.AccentTrackOff
    track.BorderSizePixel  = 0
    track.Parent           = row
    applyCorner(track, 99)

    local knob = Instance.new("Frame")
    knob.Size             = UDim2.new(0, 14, 0, 14)
    knob.Position         = state and UDim2.new(1, -16, 0.5, -7) or UDim2.new(0, 2, 0.5, -7)
    knob.BackgroundColor3 = Color3.new(1, 1, 1)
    knob.BorderSizePixel  = 0
    knob.Parent           = track
    applyCorner(knob, 99)

    local hitbox = Instance.new("TextButton")
    hitbox.Size              = UDim2.new(1, 0, 1, 0)
    hitbox.BackgroundTransparency = 1
    hitbox.Text              = ""
    hitbox.Parent            = row

    hitbox.MouseButton1Click:Connect(function()
        state                = not state
        Settings[key]        = state
        local trackColor     = state and Theme.AccentTrackOn or Theme.AccentTrackOff
        local knobPos        = state and UDim2.new(1, -16, 0.5, -7) or UDim2.new(0, 2, 0.5, -7)
        tween(track, 0.12, {BackgroundColor3 = trackColor})
        tween(knob,  0.12, {Position         = knobPos})
        if callback then pcall(callback, state) end
    end)

    return function() return state end
end

-- Draggable slider bound to Settings[key]
function UI.slider(panel, label, key, minVal, maxVal, callback)
    local value = Settings[key] or minVal

    local row = Instance.new("Frame")
    row.Size             = UDim2.new(1, -4, 0, 42)
    row.BackgroundColor3 = Theme.BgRow
    row.BorderSizePixel  = 0
    row.Parent           = panel
    applyCorner(row, 5)

    local lbl = Instance.new("TextLabel")
    lbl.Size              = UDim2.new(1, -8, 0, 18)
    lbl.Position          = UDim2.new(0, 8, 0, 4)
    lbl.BackgroundTransparency = 1
    lbl.Text              = label .. ": " .. tostring(value)
    lbl.TextColor3        = Theme.TextSecondary
    lbl.Font              = Enum.Font.Gotham
    lbl.TextSize          = 11
    lbl.TextXAlignment    = Enum.TextXAlignment.Left
    lbl.Parent            = row

    local trackBg = Instance.new("Frame")
    trackBg.Size             = UDim2.new(1, -16, 0, 6)
    trackBg.Position         = UDim2.new(0, 8, 0, 28)
    trackBg.BackgroundColor3 = Theme.TrackBg
    trackBg.BorderSizePixel  = 0
    trackBg.Parent           = row
    applyCorner(trackBg, 99)

    local fill = Instance.new("Frame")
    fill.Size             = UDim2.new((value - minVal) / (maxVal - minVal), 0, 1, 0)
    fill.BackgroundColor3 = Theme.AccentSlider
    fill.BorderSizePixel  = 0
    fill.Parent           = trackBg
    applyCorner(fill, 99)

    local handle = Instance.new("Frame")
    handle.Size             = UDim2.new(0, 12, 0, 12)
    handle.AnchorPoint      = Vector2.new(0.5, 0.5)
    handle.Position         = UDim2.new((value - minVal) / (maxVal - minVal), 0, 0.5, 0)
    handle.BackgroundColor3 = Theme.AccentHandle
    handle.BorderSizePixel  = 0
    handle.ZIndex           = 2
    handle.Parent           = trackBg
    applyCorner(handle, 99)

    local hitbox = Instance.new("TextButton")
    hitbox.Size              = UDim2.new(1, 0, 1, 10)
    hitbox.AnchorPoint       = Vector2.new(0, 0.5)
    hitbox.Position          = UDim2.new(0, 0, 0.5, 0)
    hitbox.BackgroundTransparency = 1
    hitbox.Text              = ""
    hitbox.ZIndex            = 3
    hitbox.Parent            = trackBg

    local function applyX(inputX)
        local rel    = math.clamp((inputX - trackBg.AbsolutePosition.X) / trackBg.AbsoluteSize.X, 0, 1)
        value        = math.floor(minVal + rel * (maxVal - minVal))
        Settings[key]        = value
        fill.Size            = UDim2.new(rel, 0, 1, 0)
        handle.Position      = UDim2.new(rel, 0, 0.5, 0)
        lbl.Text             = label .. ": " .. tostring(value)
        if callback then pcall(callback, value) end
    end

    hitbox.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            activeDrag = applyX
            applyX(input.Position.X)
        end
    end)

    return function() return value end
end

-- ESP ─────────────────────────────────────────────────────────────────────────

local espObjects = {}

function Features.clearESP()
    for _, obj in pairs(espObjects) do
        pcall(function()
            if obj and obj.Parent then obj:Destroy() end
        end)
    end
    espObjects = {}
end

function Features.buildESP(player)
    if player == LocalPlayer then return end
    if Settings.ESPTeamCheck and player.Team == LocalPlayer.Team then return end
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum  = char:FindFirstChildOfClass("Humanoid")
    if not root then return end

    if Settings.ESPBoxes then
        local box            = Instance.new("SelectionBox")
        box.Adornee          = char
        box.Color3           = Color3.fromRGB(255, 50, 50)
        box.LineThickness    = 0.04
        box.SurfaceTransparency = 0.8
        box.SurfaceColor3    = Color3.fromRGB(255, 50, 50)
        box.Parent           = workspace
        espObjects[player.Name .. "_box"] = box
    end

    if Settings.ESPNames or Settings.ESPHealth or Settings.ESPDistance then
        local bb = Instance.new("BillboardGui")
        bb.Adornee     = root
        bb.AlwaysOnTop = true
        bb.Size        = UDim2.new(0, 100, 0, 40)
        bb.StudsOffset = Vector3.new(0, 3.5, 0)
        bb.Parent      = workspace

        if Settings.ESPNames then
            local nameLbl                  = Instance.new("TextLabel")
            nameLbl.Text                   = player.Name
            nameLbl.Size                   = UDim2.new(1, 0, 0.5, 0)
            nameLbl.BackgroundTransparency = 1
            nameLbl.TextColor3             = Color3.fromRGB(255, 60, 60)
            nameLbl.Font                   = Enum.Font.GothamBold
            nameLbl.TextSize               = 13
            nameLbl.TextStrokeTransparency = 0.4
            nameLbl.Parent                 = bb
        end

        if Settings.ESPHealth and hum then
            local hpLbl                  = Instance.new("TextLabel")
            hpLbl.Text                   = string.format("HP: %d/%d",
                math.floor(hum.Health), math.floor(hum.MaxHealth))
            hpLbl.Size                   = UDim2.new(1, 0, 0.5, 0)
            hpLbl.Position               = UDim2.new(0, 0, 0.5, 0)
            hpLbl.BackgroundTransparency = 1
            hpLbl.TextColor3             = Color3.fromRGB(80, 220, 80)
            hpLbl.Font                   = Enum.Font.Gotham
            hpLbl.TextSize               = 11
            hpLbl.TextStrokeTransparency = 0.4
            hpLbl.Parent                 = bb
        end

        espObjects[player.Name .. "_bb"] = bb
    end
end

-- Fly ─────────────────────────────────────────────────────────────────────────

local flyVelocity, flyGyro

function Features.stopFly()
    if flyVelocity and flyVelocity.Parent then flyVelocity:Destroy() end
    if flyGyro     and flyGyro.Parent     then flyGyro:Destroy()     end
    flyVelocity = nil
    flyGyro     = nil
    if Humanoid then Humanoid.PlatformStand = false end
end

function Features.tickFly()
    if not (Settings.FlyOn and RootPart and Humanoid) then return end

    if not flyVelocity or not flyVelocity.Parent then
        flyVelocity          = Instance.new("BodyVelocity")
        flyVelocity.MaxForce = Vector3.new(1e9, 1e9, 1e9)
        flyVelocity.Velocity = Vector3.new()
        flyVelocity.Parent   = RootPart
    end
    if not flyGyro or not flyGyro.Parent then
        flyGyro           = Instance.new("BodyGyro")
        flyGyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
        flyGyro.P         = 1e4
        flyGyro.Parent    = RootPart
    end

    Humanoid.PlatformStand = true

    local UIS = UserInputService
    local cf  = Camera.CFrame
    local dir = Vector3.new()

    if UIS:IsKeyDown(Enum.KeyCode.W) then dir += cf.LookVector  end
    if UIS:IsKeyDown(Enum.KeyCode.S) then dir -= cf.LookVector  end
    if UIS:IsKeyDown(Enum.KeyCode.A) then dir -= cf.RightVector end
    if UIS:IsKeyDown(Enum.KeyCode.D) then dir += cf.RightVector end
    if UIS:IsKeyDown(Enum.KeyCode.Space)       then dir += Vector3.yAxis end
    if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= Vector3.yAxis end

    local speed = Settings.FlySpeed * (UIS:IsKeyDown(Enum.KeyCode.LeftShift) and 3 or 1)
    flyVelocity.Velocity = dir.Magnitude > 0 and dir.Unit * speed or Vector3.new()
    flyGyro.CFrame       = cf
end

-- Aimbot ──────────────────────────────────────────────────────────────────────

function Features.getNearestTarget()
    local screenCenter = Camera.ViewportSize / 2
    local nearest, nearestDist = nil, Settings.AimbotFOV

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer or not plr.Character then continue end
        if Settings.AimbotTeamCheck and plr.Team == LocalPlayer.Team then continue end

        local hitPart = plr.Character:FindFirstChild(Settings.AimbotHitPart)
            or plr.Character:FindFirstChild("HumanoidRootPart")
        if not hitPart then continue end

        local screenPos, onScreen = Camera:WorldToScreenPoint(hitPart.Position)
        if onScreen then
            local dist = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
            if dist < nearestDist then
                nearestDist = dist
                nearest     = hitPart
            end
        end
    end

    return nearest
end

-- Anti-Slip ───────────────────────────────────────────────────────────────────

local SLIPPERY_PROPS = PhysicalProperties.new(0.3, 0, 0, 0, 0)
local DEFAULT_PROPS  = PhysicalProperties.new(0.7, 0.3, 0.5, 0, 0)

function Features.setAntiSlip(state)
    if not Character then return end
    local props = state and SLIPPERY_PROPS or DEFAULT_PROPS
    for _, part in ipairs(Character:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CustomPhysicalProperties = props
        end
    end
end

-- ─────────────────────────────────────────────────────────────────────────────
--  CHARACTER ADDED HOOK
-- ─────────────────────────────────────────────────────────────────────────────

local function onCharacterAdded(newChar)
    syncCharacter(newChar)

    if Settings.GodModeOn then
        Humanoid.MaxHealth = math.huge
        Humanoid.Health    = math.huge
    end

    if Settings.NoAnimOn then
        local animator = Humanoid:FindFirstChildOfClass("Animator")
        if animator then
            for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
                track:Stop(0)
            end
        end
    end

    if Settings.InvisibleOn then
        for _, part in ipairs(newChar:GetDescendants()) do
            if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                part.Transparency = 1
            end
        end
    end

    if Settings.AntiSlip then
        Features.setAntiSlip(true)
    end
end

connect("charAdded", LocalPlayer.CharacterAdded, onCharacterAdded)

local tabButtons = {}
local tabPanels  = {}

local function switchTab(name)
    for n, btn in pairs(tabButtons) do
        btn.BackgroundColor3 = n == name and Theme.TabActive   or Theme.TabInactive
        btn.TextColor3       = n == name and Theme.TextTabOn   or Theme.TextTabOff
    end
    for n, panel in pairs(tabPanels) do
        panel.Visible = (n == name)
    end
end

local tabOrder = 0
local function addTab(name)
    tabOrder = tabOrder + 1
    local panel = UI.addTab(Sidebar, ContentArea, tabButtons, tabPanels, name, tabOrder)
    tabButtons[name].MouseButton1Click:Connect(function() switchTab(name) end)
    return panel
end

local function executeScript(src)
    if not src or #src == 0 then return end
    local fn, compileErr = loadstring(src)
    if not fn then
        UI.notify("Syntax Error", tostring(compileErr), 4)
        return
    end
    local ok, runErr = pcall(fn)
    if not ok then
        UI.notify("Runtime Error", tostring(runErr), 4)
    end
end

-- ─────────────────────────────────────────────────────────────────────────────
--  RUNTIME LOOPS
-- ─────────────────────────────────────────────────────────────────────────────

local fpsFrames  = 0
local fpsLastTick = tick()

connect("renderStep", RunService.RenderStepped, function()
    -- FPS counter
    fpsFrames += 1
    local now = tick()
    if now - fpsLastTick >= 1 then
        fpsCounter.Visible = Settings.ShowFPS
        if Settings.ShowFPS then
            fpsCounter.Text = "FPS: " .. fpsFrames
        end
        fpsFrames  = 0
        fpsLastTick = now
    end

    -- Fly
    Features.tickFly()
end)

connect("stepped", RunService.Stepped, function()
    -- Noclip
    if Settings.NoclipOn and Character then
        for _, part in ipairs(Character:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end

    -- Anti-void
    if Settings.AntiVoidOn and RootPart and RootPart.Position.Y < -150 then
        RootPart.CFrame = CFrame.new(0, 10, 0)
    end

    -- Speed hack
    if Settings.SpeedHackOn and Humanoid then
        Humanoid.WalkSpeed = Settings.WalkSpeed * Settings.SpeedMult
    end
end)

connect("heartbeat", RunService.Heartbeat, function()
    -- ESP
    if Settings.ESPOn then
        Features.clearESP()
        for _, plr in ipairs(Players:GetPlayers()) do
            pcall(Features.buildESP, plr)
        end
    end

    -- God mode
    if Settings.GodModeOn and Humanoid then
        Humanoid.MaxHealth = math.huge
        Humanoid.Health    = math.huge
    end

    -- Auto respawn
    if Settings.AutoRespawn and Humanoid
    and Humanoid:GetState() == Enum.HumanoidStateType.Dead then
        task.wait(0.5)
        LocalPlayer:LoadCharacter()
    end

    -- Aimbot
    if Settings.AimbotOn then
        local target = Features.getNearestTarget()
        if target then
            Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, target.Position)
        end
    end
end)

connect("infiniteJump", UserInputService.JumpRequest, function()
    if Settings.InfJumpOn and Humanoid then
        Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
    end
end)
